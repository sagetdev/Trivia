import { Router, type Request, type Response } from "express";
import { randomUUID } from "node:crypto";
import {
  getCategories,
  getQuestions,
  type QuizQuestion,
  type Difficulty,
  type QuestionType,
} from "../services/opentdb.service";

const router = Router();

interface QuizSession {
  id: string;
  playerName: string;
  questions: QuizQuestion[];
  currentIndex: number;
  score: number;
  answeredCurrent: boolean;
  finishedAt?: string;
}

interface LeaderboardEntry {
  quizId: string;
  playerName: string;
  score: number;
  total: number;
  finishedAt: string;
}

const sessions = new Map<string, QuizSession>();
const leaderboard: LeaderboardEntry[] = [];

function publicQuestion(question: QuizQuestion) {
  return {
    category: question.category,
    difficulty: question.difficulty,
    type: question.type,
    question: question.question,
    answers: question.answers,
  };
}

function getSession(id: string, res: Response): QuizSession | undefined {
  const session = sessions.get(id);

  if (!session) {
    res.status(404).json({ error: "Partie introuvable." });
    return undefined;
  }

  return session;
}

function currentQuestion(session: QuizSession) {
  const question = session.questions[session.currentIndex];
  return question ? publicQuestion(question) : null;
}

router.get("/categories", async (_req, res) => {
  try {
    const categories = await getCategories();
    res.json({ categories });
  } catch (error) {
    res.status(502).json({
      error: error instanceof Error ? error.message : "Erreur OpenTDB.",
    });
  }
});

router.post("/quizzes", async (req, res) => {
  const body = req.body as Record<string, unknown>;

  const amount = body.amount ?? 10;
  const categoryId = body.categoryId;
  const difficulty = body.difficulty;
  const type = body.type;
  const playerName = body.playerName ?? "Joueur";

  if (
    typeof amount !== "number" ||
    !Number.isInteger(amount) ||
    amount < 1 ||
    amount > 20
  ) {
    res.status(400).json({ error: "amount doit être entre 1 et 20." });
    return;
  }

  if (
    categoryId !== undefined &&
    (typeof categoryId !== "number" ||
      !Number.isInteger(categoryId) ||
      categoryId < 1)
  ) {
    res.status(400).json({ error: "categoryId invalide." });
    return;
  }

  if (
    difficulty !== undefined &&
    !["easy", "medium", "hard"].includes(String(difficulty))
  ) {
    res.status(400).json({ error: "difficulty invalide." });
    return;
  }

  if (
    type !== undefined &&
    !["multiple", "boolean"].includes(String(type))
  ) {
    res.status(400).json({ error: "type invalide." });
    return;
  }

  if (
    typeof playerName !== "string" ||
    playerName.trim().length < 1 ||
    playerName.trim().length > 30
  ) {
    res.status(400).json({ error: "playerName doit contenir 1 à 30 caractères." });
    return;
  }

  try {
    const questions = await getQuestions({
      amount,
      categoryId: categoryId as number | undefined,
      difficulty: difficulty as Difficulty | undefined,
      type: type as QuestionType | undefined,
    });

    const id = randomUUID();

    const session: QuizSession = {
      id,
      playerName: playerName.trim(),
      questions,
      currentIndex: 0,
      score: 0,
      answeredCurrent: false,
    };

    sessions.set(id, session);

    res.status(201).json({
      quizId: id,
      totalQuestions: questions.length,
      question: publicQuestion(questions[0]),
    });
  } catch (error) {
    res.status(502).json({
      error: error instanceof Error ? error.message : "Impossible de créer la partie.",
    });
  }
});

router.get("/quizzes/:id", (req, res) => {
  const session = getSession(req.params.id, res);
  if (!session) return;

  res.json({
    quizId: session.id,
    currentIndex: session.currentIndex,
    totalQuestions: session.questions.length,
    score: session.score,
    finished: Boolean(session.finishedAt),
    question: currentQuestion(session),
  });
});

router.post("/quizzes/:id/answers", (req, res) => {
  const session = getSession(req.params.id, res);
  if (!session) return;

  if (session.finishedAt) {
    res.status(409).json({ error: "Cette partie est terminée." });
    return;
  }

  if (session.answeredCurrent) {
    res.status(409).json({ error: "Tu as déjà répondu à cette question." });
    return;
  }

  const answer = req.body?.answer;

  if (typeof answer !== "string") {
    res.status(400).json({ error: "answer doit être une chaîne de caractères." });
    return;
  }

  const question = session.questions[session.currentIndex];

  if (!question) {
    res.status(409).json({ error: "Aucune question disponible." });
    return;
  }

  const isCorrect = answer === question.correctAnswer;

  session.answeredCurrent = true;

  if (isCorrect) {
    session.score += 1;
  }

  const answeredIndex = session.currentIndex;

  // La bonne réponse n'est révélée qu'après la soumission.
  session.currentIndex += 1;
  session.answeredCurrent = false;

  const finished = session.currentIndex >= session.questions.length;

  if (finished) {
    session.finishedAt = new Date().toISOString();

    leaderboard.push({
      quizId: session.id,
      playerName: session.playerName,
      score: session.score,
      total: session.questions.length,
      finishedAt: session.finishedAt,
    });

    leaderboard.sort(
      (a, b) => b.score / b.total - a.score / a.total || b.score - a.score,
    );

    leaderboard.splice(100);
  }

  res.json({
    isCorrect,
    correctAnswer: question.correctAnswer,
    answeredIndex,
    score: session.score,
    totalQuestions: session.questions.length,
    finished,
    nextQuestion: finished ? null : currentQuestion(session),
    ...(finished
      ? {
          result: {
            quizId: session.id,
            playerName: session.playerName,
            score: session.score,
            total: session.questions.length,
            finishedAt: session.finishedAt,
          },
        }
      : {}),
  });
});

router.get("/quizzes/:id/results", (req, res) => {
  const session = getSession(req.params.id, res);
  if (!session) return;

  if (!session.finishedAt) {
    res.status(409).json({ error: "La partie n'est pas terminée." });
    return;
  }

  res.json({
    quizId: session.id,
    playerName: session.playerName,
    score: session.score,
    total: session.questions.length,
    finishedAt: session.finishedAt,
  });
});

router.get("/leaderboard", (_req, res) => {
  res.json({
    leaderboard: leaderboard.map((entry, index) => ({
      rank: index + 1,
      ...entry,
    })),
  });
});

export default router;

===== \Backend\src\services\opentdb.service.ts =====

import he from "he";

const BASE_URL = "https://opentdb.com";

export type Difficulty = "easy" | "medium" | "hard";
export type QuestionType = "multiple" | "boolean";

export interface QuizQuestion {
  category: string;
  difficulty: Difficulty;
  type: QuestionType;
  question: string;
  correctAnswer: string;
  answers: string[];
}

interface OpenTDBCategory {
  id: number;
  name: string;
}

interface OpenTDBQuestion {
  category: string;
  difficulty: Difficulty;
  type: QuestionType;
  question: string;
  correct_answer: string;
  incorrect_answers: string[];
}

interface OpenTDBResponse<T> {
  response_code: number;
  results?: T[];
  trivia_categories?: OpenTDBCategory[];
}

let categoriesCache:
  | { expiresAt: number; data: OpenTDBCategory[] }
  | undefined;

let lastQuestionRequest = 0;
let questionQueue: Promise<void> = Promise.resolve();

function decode(value: string): string {
  return he.decode(value);
}

function shuffle<T>(items: T[]): T[] {
  const result = [...items];

  for (let i = result.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [result[i], result[j]] = [result[j], result[i]];
  }

  return result;
}

async function request<T>(url: string): Promise<T> {
  const response = await fetch(url, {
    headers: { Accept: "application/json" },
    signal: AbortSignal.timeout(10000),
  });

  if (!response.ok) {
    throw new Error(`OpenTDB a répondu HTTP ${response.status}`);
  }

  return (await response.json()) as T;
}

export async function getCategories(): Promise<OpenTDBCategory[]> {
  if (categoriesCache && categoriesCache.expiresAt > Date.now()) {
    return categoriesCache.data;
  }

  const data = await request<OpenTDBResponse<never>>(
    `${BASE_URL}/api_category.php`,
  );

  if (data.response_code !== 0 || !data.trivia_categories) {
    throw new Error("Impossible de récupérer les catégories OpenTDB.");
  }

  categoriesCache = {
    data: data.trivia_categories,
    expiresAt: Date.now() + 5 * 60 * 1000,
  };

  return data.trivia_categories;
}

async function waitForQuestionSlot(): Promise<void> {
  const task = questionQueue.then(async () => {
    const wait = Math.max(0, 5000 - (Date.now() - lastQuestionRequest));

    if (wait > 0) {
      await new Promise<void>((resolve) => setTimeout(resolve, wait));
    }

    lastQuestionRequest = Date.now();
  });

  questionQueue = task.catch(() => {});
  await task;
}

export async function getQuestions(options: {
  amount: number;
  categoryId?: number;
  difficulty?: Difficulty;
  type?: QuestionType;
}): Promise<QuizQuestion[]> {
  await waitForQuestionSlot();

  const params = new URLSearchParams({
    amount: String(options.amount),
    encode: "default",
  });

  if (options.categoryId !== undefined) {
    params.set("category", String(options.categoryId));
  }

  if (options.difficulty) {
    params.set("difficulty", options.difficulty);
  }

  if (options.type) {
    params.set("type", options.type);
  }

  const data = await request<OpenTDBResponse<OpenTDBQuestion>>(
    `${BASE_URL}/api.php?${params.toString()}`,
  );

  if (data.response_code === 1) {
    throw new Error(
      "Pas assez de questions pour ces filtres. Essaie une autre catégorie ou difficulté.",
    );
  }

  if (data.response_code === 5) {
    throw new Error("OpenTDB limite les requêtes. Réessaie dans quelques secondes.");
  }

  if (data.response_code !== 0 || !data.results) {
    throw new Error(`Réponse OpenTDB invalide (code ${data.response_code}).`);
  }

  return data.results.map((item) => {
    const correctAnswer = decode(item.correct_answer);
    const answers = shuffle([
      correctAnswer,
      ...item.incorrect_answers.map(decode),
    ]);

    return {
      category: decode(item.category),
      difficulty: item.difficulty,
      type: item.type,
      question: decode(item.question),
      correctAnswer,
      answers,
    };
  });
}

import React, { useCallback, useMemo, useRef, useState } from 'react';
import { ScrollView, Text as RNText, View } from 'react-native';
import { useLocalSearchParams, useRouter } from 'expo-router';
import { SafeAreaView, useSafeAreaInsets } from 'react-native-safe-area-context';
import { colors, fonts } from '../../theme';
import { getCategory, getQuestions } from '../../components/data/mock';
import { formatTime } from '../../components/data/format';
import { useCountdown } from '../../hooks/useCountdown';
import { Button, CategoryTag, OptionCard, OptionState, ProgressBar } from '../../components/ui';
import { FeedbackSheet, QuizTopBar, Verdict } from '../../components/features';

const LETTERS = ['A', 'B', 'C', 'D'];
const LIFELINES = 2;
const XP_PER_QUESTION: Record<string, number> = { easy: 8, medium: 12, hard: 16 };

export default function QuizPlayScreen() {
  const router = useRouter();
  const insets = useSafeAreaInsets();
  const params = useLocalSearchParams<{ category?: string; difficulty?: string; count?: string; time?: string }>();

  const category = getCategory(params.category);
  const difficulty = params.difficulty ?? 'medium';
  const total = Number(params.count) || 10;
  const timeLimit = params.time === 'none' ? null : Number(params.time) || 30;

  const questions = useMemo(() => getQuestions(total), [total]);
  const [index, setIndex] = useState(0);
  const [selected, setSelected] = useState<number | null>(null);
  const [revealed, setRevealed] = useState(false);
  const [removed, setRemoved] = useState<number[]>([]);
  const [lifelines, setLifelines] = useState(LIFELINES);
  const [timedOut, setTimedOut] = useState(false);

  const correctCount = useRef(0);
  const startedAt = useRef(Date.now());

  const question = questions[index];
  const isLast = index === questions.length - 1;

  const reveal = useCallback(
    (choice: number | null) => {
      if (choice !== null && choice === question.correctIndex) correctCount.current += 1;
      setRevealed(true);
    },
    [question],
  );

  const secondsLeft = useCountdown(timeLimit, !revealed, index, () => {
    setTimedOut(true);
    reveal(null);
  });

  const submit = () => {
    if (selected === null) return;
    reveal(selected);
  };

  const applyLifeline = () => {
    if (lifelines === 0 || revealed) return;
    const wrong = question.options
      .map((_, i) => i)
      .filter((i) => i !== question.correctIndex && i !== selected);
    setRemoved(wrong.sort(() => Math.random() - 0.5).slice(0, 2));
    setLifelines((n) => n - 1);
  };

  const next = () => {
    if (isLast) {
      router.replace({
        pathname: '/(quiz)/results',
        params: {
          category: category.id,
          score: String(correctCount.current),
          total: String(questions.length),
          seconds: String(Math.round((Date.now() - startedAt.current) / 1000)),
          xp: String(correctCount.current * (XP_PER_QUESTION[difficulty] ?? 12)),
        },
      });
      return;
    }
    setIndex((i) => i + 1);
    setSelected(null);
    setRevealed(false);
    setRemoved([]);
    setTimedOut(false);
  };

  const stateFor = (i: number): OptionState => {
    if (!revealed) return selected === i ? 'selected' : 'default';
    if (i === question.correctIndex) return 'correct';
    if (i === selected) return 'wrong';
    return 'default';
  };

  const verdict: Verdict = !revealed
    ? 'wrong'
    : timedOut
      ? 'timeout'
      : selected === question.correctIndex
        ? 'correct'
        : 'wrong';

  return (
    <SafeAreaView edges={['top']} style={{ flex: 1, backgroundColor: colors.bg }}>
      <ScrollView
        showsVerticalScrollIndicator={false}
        contentContainerStyle={{ paddingHorizontal: 20, paddingTop: 8, paddingBottom: 24, gap: 20 }}
      >
        <QuizTopBar
          title={`Question ${index + 1} of ${questions.length}`}
          onClose={() => router.replace('/home')}
          timer={timeLimit === null ? null : formatTime(secondsLeft)}
        />

        <ProgressBar progress={(index + 1) / questions.length} />

        <CategoryTag
          icon={category.icon}
          color={category.color}
          label={`${category.name} · ${difficulty.charAt(0).toUpperCase()}${difficulty.slice(1)}`}
        />

        <RNText
          style={{ fontFamily: fonts.bold, fontSize: 26, lineHeight: 34, color: colors.text, includeFontPadding: false }}
        >
          {question.text}
        </RNText>

        <View style={{ gap: 12 }}>
          {question.options.map((option, i) => (
            <OptionCard
              key={option}
              letter={LETTERS[i]}
              label={option}
              state={stateFor(i)}
              disabled={revealed || removed.includes(i)}
              dimmed={removed.includes(i) && !revealed}
              onPress={() => setSelected(i)}
            />
          ))}
        </View>
      </ScrollView>

      {revealed ? (
        <FeedbackSheet
          verdict={verdict}
          answer={question.options[question.correctIndex]}
          explanation={question.explanation}
          isLast={isLast}
          onNext={next}
        />
      ) : (
        <View style={{ paddingHorizontal: 20, paddingBottom: Math.max(insets.bottom, 20) + 14, gap: 12 }}>
          <Button
            label={`Use 50:50 lifeline (${lifelines} left)`}
            variant="soft"
            disabled={lifelines === 0 || removed.length > 0}
            onPress={applyLifeline}
          />
          <Button label="Submit answer" disabled={selected === null} onPress={submit} />
        </View>
      )}
    </SafeAreaView>
  );
}

import React from 'react';
import { ScrollView, View } from 'react-native';
import { useLocalSearchParams, useRouter } from 'expo-router';
import { SafeAreaView, useSafeAreaInsets } from 'react-native-safe-area-context';
import { colors } from '../../theme';
import { getCategory } from '../../components/data/mock';
import { formatTime } from '../../components/data/format';
import { Button, Card, Icon, IconButton, IconName, ScoreRing, Text } from '../../components/ui';

const message = (ratio: number) => {
  if (ratio >= 0.8) return 'Great job!';
  if (ratio >= 0.5) return 'Nice effort!';
  return 'Keep practicing!';
};

export default function QuizResultsScreen() {
  const router = useRouter();
  const insets = useSafeAreaInsets();
  const params = useLocalSearchParams<{ category?: string; score?: string; total?: string; seconds?: string; xp?: string }>();

  const category = getCategory(params.category);
  const score = Number(params.score) || 0;
  const total = Number(params.total) || 10;
  const seconds = Number(params.seconds) || 0;
  const xp = Number(params.xp) || 0;
  const percentile = Math.min(99, Math.round((score / total) * 95));

  const stats: { icon: IconName; color: string; value: string; label: string }[] = [
    { icon: 'check', color: colors.success, value: String(score), label: 'Correct' },
    { icon: 'clock', color: colors.warning, value: formatTime(seconds), label: 'Time' },
    { icon: 'zap', color: colors.primary, value: `+${xp}`, label: 'XP earned' },
  ];

  return (
    <SafeAreaView edges={['top']} style={{ flex: 1, backgroundColor: colors.bg }}>
      <ScrollView
        showsVerticalScrollIndicator={false}
        contentContainerStyle={{
          flexGrow: 1,
          justifyContent: 'space-between',
          paddingHorizontal: 20,
          paddingTop: 8,
          paddingBottom: Math.max(insets.bottom, 20) + 14,
          gap: 24,
        }}
      >
        <View style={{ flexDirection: 'row', alignItems: 'center', justifyContent: 'space-between' }}>
          <IconButton icon="x" label="Close" onPress={() => router.replace('/home')} />
          <Text size={15} weight="semibold" color={colors.textMuted}>
            Quiz complete
          </Text>
          <View style={{ width: 44 }} />
        </View>

        <View style={{ alignItems: 'center', gap: 24 }}>
          <ScoreRing score={score} total={total} />

          <View style={{ alignItems: 'center', gap: 8, width: '100%' }}>
            <Text size={32} weight="bold">
              {message(score / total)}
            </Text>
            <Text size={16} lineHeight={24} color={colors.textMuted} align="center">
              You scored higher than {percentile}% of players on {category.name} today.
            </Text>
          </View>

          <View style={{ flexDirection: 'row', gap: 12, width: '100%' }}>
            {stats.map((s) => (
              <Card key={s.label} radius={16} style={{ flex: 1, alignItems: 'center', gap: 4, padding: 14 }}>
                <Icon name={s.icon} size={20} color={s.color} />
                <Text size={18} weight="semibold">
                  {s.value}
                </Text>
                <Text size={12} color={colors.textMuted}>
                  {s.label}
                </Text>
              </Card>
            ))}
          </View>
        </View>

        <View style={{ gap: 8 }}>
          <Button
            label="Play again"
            onPress={() => router.replace({ pathname: '/quiz/setup', params: { category: category.id } })}
          />
          <Button label="Review answers" variant="secondary" />

export const formatTime = (totalSeconds: number) => {
  const m = Math.floor(totalSeconds / 60);
  const s = Math.floor(totalSeconds % 60);
  return `${m}:${s.toString().padStart(2, '0')}`;
};

import React from 'react';
import { View } from 'react-native';
import Svg, { Circle } from 'react-native-svg';
import { colors } from '../../theme';
import { Text } from './Text';

interface Props {
  score: number;
  total: number;
  size?: number;
  stroke?: number;
}

export function ScoreRing({ score, total, size = 200, stroke = 14 }: Props) {
  const r = (size - stroke) / 2;
  const c = 2 * Math.PI * r;
  const progress = total > 0 ? score / total : 0;
  const cx = size / 2;

  return (
    <View style={{ width: size, height: size, alignItems: 'center', justifyContent: 'center', gap: 2 }}>
      <Svg width={size} height={size} style={{ position: 'absolute' }}>
        <Circle cx={cx} cy={cx} r={r} stroke={colors.surfaceAlt} strokeWidth={stroke} fill="none" />
        <Circle
          cx={cx}
          cy={cx}
          r={r}
          stroke={colors.primary}
          strokeWidth={stroke}
          fill="none"
          strokeLinecap="round"
          strokeDasharray={`${c} ${c}`}
          strokeDashoffset={c * (1 - progress)}
          rotation={-90}
          origin={`${cx}, ${cx}`}
        />
      </Svg>
      <Text size={48} weight="bold">
        {score}/{total}
      </Text>
      <Text size={13} weight="medium" color={colors.textMuted}>
        Correct answers
      </Text>
    </View>
  );
}

import React, { useEffect, useRef } from 'react';
import { Animated, Pressable } from 'react-native';
import { colors } from '../../theme';

interface Props {
  value: boolean;
  onValueChange: (value: boolean) => void;
}

export function Toggle({ value, onValueChange }: Props) {
  const x = useRef(new Animated.Value(value ? 1 : 0)).current;

  useEffect(() => {
    Animated.timing(x, { toValue: value ? 1 : 0, duration: 160, useNativeDriver: false }).start();
  }, [value, x]);

  const bg = x.interpolate({ inputRange: [0, 1], outputRange: [colors.border, colors.primary] });
  const left = x.interpolate({ inputRange: [0, 1], outputRange: [2, 22] });

  return (
    <Pressable
      accessibilityRole="switch"
      accessibilityState={{ checked: value }}
      onPress={() => onValueChange(!value)}
      hitSlop={8}
    >
      <Animated.View style={{ width: 46, height: 26, borderRadius: 13, backgroundColor: bg }}>
        <Animated.View
          style={{
            position: 'absolute',
            top: 2,
            left,
            width: 22,
            height: 22,
            borderRadius: 11,
            backgroundColor: colors.onPrimary,
          }}

===== \Frontend\src\hooks\useCountdown.ts =====

import { useEffect, useState } from 'react';

export function useCountdown(seconds: number | null, running: boolean, resetKey: unknown, onEnd: () => void) {
  const [left, setLeft] = useState(seconds ?? 0);

  useEffect(() => {
    setLeft(seconds ?? 0);
  }, [seconds, resetKey]);

  useEffect(() => {
    if (seconds === null || !running) return;
    const id = setInterval(() => {
      setLeft((prev) => {
        if (prev <= 1) {
          clearInterval(id);
          onEnd();
          return 0;
        }
        return prev - 1;
      });
    }, 1000);
    return () => clearInterval(id);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [seconds, running, resetKey]);

  return left;
}

const API_BASE_URL = (
  process.env.EXPO_PUBLIC_API_URL ?? "http://10.0.2.2:4000/api"
).replace(/\/+$/, "");

export interface QuizCategory {
  id: number;
  name: string;
}

export interface QuizQuestion {
  category: string;
  difficulty: "easy" | "medium" | "hard";
  type: "multiple" | "boolean";
  question: string;
  answers: string[];
}

export interface QuizSession {
  quizId: string;
  totalQuestions: number;
  question: QuizQuestion | null;
}

export interface QuizResult {
  correct: boolean;
  correctAnswer: string;
  score: number;
  totalQuestions?: number;
  finished?: boolean;
  nextQuestion?: QuizQuestion | null;
}

export interface CreateQuizOptions {
  amount: number;
  categoryId?: number;
  difficulty?: "easy" | "medium" | "hard";
  type?: "multiple" | "boolean";
  playerName: string;
}

async function request<T>(
  path: string,
  options: RequestInit = {},
): Promise<T> {
  let response: Response;

  try {
    response = await fetch(`${API_BASE_URL}${path}`, {
      ...options,
      headers: {
        Accept: "application/json",
        "Content-Type": "application/json",
        ...options.headers,
      },
    });
  } catch {
    throw new Error(
      `Impossible de joindre le serveur (${API_BASE_URL}). Vérifie que le backend est démarré et que son adresse est correcte.`,
    );
  }

  const data = await response.json().catch(() => ({}));

  if (!response.ok) {
    throw new Error(
      typeof data?.error === "string"
        ? data.error
        : `Erreur API HTTP ${response.status}`,
    );
  }

  return data as T;
}

export const api = {
  health: () =>
    request<{ status: string; service: string }>("/health"),

  getCategories: () =>
    request<QuizCategory[]>("/categories"),

  createQuiz: (options: CreateQuizOptions) =>
    request<QuizSession>("/quizzes", {
      method: "POST",
      body: JSON.stringify(options),
    }),

  getQuiz: (quizId: string) =>
    request<QuizSession & { score: number; finished: boolean }>(
      `/quizzes/${encodeURIComponent(quizId)}`,
    ),

  submitAnswer: (quizId: string, answer: string) =>
    request<QuizResult>(
      `/quizzes/${encodeURIComponent(quizId)}/answers`,
      {
        method: "POST",
        body: JSON.stringify({ answer }),
      },
    ),

  getResults: (quizId: string) =>
    request<QuizResult>(
      `/quizzes/${encodeURIComponent(quizId)}/results`,
    ),

  getLeaderboard: () =>
    request<Array<{
      playerName: string;
      score: number;
      totalQuestions: number;
    }>>("/leaderboard"),
};
